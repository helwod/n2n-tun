# CMakeLists.txt - CMake build configuration for n2n
# Supports Windows (MinGW), Linux, macOS, FreeBSD, NetBSD
#
# Windows build with wintun support:
#   cmake -B build -G "MinGW Makefiles" -DCMAKE_TOOLCHAIN_FILE=cmake/mingw-toolchain.cmake
#   cmake --build build
#
# Linux build:
#   cmake -B build
#   cmake --build build

cmake_minimum_required(VERSION 3.16)
project(n2n C)

# Set C standard
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Options
option(N2N_ENABLE_WINTUN "Enable wintun driver support on Windows" ON)
option(N2N_ENABLE_OPENSSL "Enable OpenSSL for encryption" ON)
option(N2N_ENABLE_ZSTD "Enable Zstd compression" ON)
option(N2N_ENABLE_TESTS "Build test tools" ON)

# Version information
set(N2N_VERSION_MAJOR 3)
set(N2N_VERSION_MINOR 1)
set(N2N_VERSION_PATCH 0)
set(N2N_VERSION "${N2N_VERSION_MAJOR}.${N2N_VERSION_MINOR}.${N2N_VERSION_PATCH}")

# Source files for n2n library
set(N2N_LIB_SOURCES
    src/aes.c
    src/auth.c
    src/cc20.c
    src/curve25519.c
    src/edge_management.c
    src/edge_utils.c
    src/header_encryption.c
    src/hexdump.c
    src/json.c
    src/management.c
    src/minilzo.c
    src/n2n.c
    src/n2n_port_mapping.c
    src/n2n_regex.c
    src/network_traffic_filter.c
    src/pearson.c
    src/random_numbers.c
    src/sn_management.c
    src/sn_selection.c
    src/sn_utils.c
    src/speck.c
    src/tf.c
    src/transform_aes.c
    src/transform_cc20.c
    src/transform_lzo.c
    src/transform_null.c
    src/transform_speck.c
    src/transform_tf.c
    src/transform_zstd.c
    src/wire.c
)

# Platform-specific TUN/TAP implementations
if(WIN32)
    # Windows platform
    set(N2N_LIB_SOURCES ${N2N_LIB_SOURCES}
        src/win32/wintap.c
        src/win32/edge_utils_win32.c
        src/win32/getopt.c
        src/win32/getopt1.c
    )

    # Add wintun support if enabled
    if(N2N_ENABLE_WINTUN)
        set(N2N_LIB_SOURCES ${N2N_LIB_SOURCES}
            src/win32/wintun_device.c
        )
        add_definitions(-DN2N_HAVE_WINTUN=1)
    endif()

    # Windows resource file
    if(EXISTS "${CMAKE_SOURCE_DIR}/src/win32/edge.rc")
        set(N2N_LIB_SOURCES ${N2N_LIB_SOURCES}
            src/win32/edge.rc
        )
    endif()
elseif(UNIX)
    if(APPLE)
        set(N2N_LIB_SOURCES ${N2N_LIB_SOURCES}
            src/tuntap_osx.c
        )
    elseif(CMAKE_SYSTEM_NAME MATCHES "Linux")
        set(N2N_LIB_SOURCES ${N2N_LIB_SOURCES}
            src/tuntap_linux.c
        )
    elseif(CMAKE_SYSTEM_NAME MATCHES "FreeBSD")
        set(N2N_LIB_SOURCES ${N2N_LIB_SOURCES}
            src/tuntap_freebsd.c
        )
    elseif(CMAKE_SYSTEM_NAME MATCHES "NetBSD")
        set(N2N_LIB_SOURCES ${N2N_LIB_SOURCES}
            src/tuntap_netbsd.c
        )
    endif()
endif()

# Include directories
include_directories(
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/src
    ${CMAKE_SOURCE_DIR}/thirdparty
    ${CMAKE_BINARY_DIR}  # For config.h
)

# Define config.h macros for Windows build
set(N2N_CONFIG_H "/* Generated by CMake */\n")
if(WIN32)
    # Create include directory in build tree
    file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/include)
    set(N2N_CONFIG_H "${N2N_CONFIG_H}
#define PACKAGE_NAME \"n2n\"
#define PACKAGE_VERSION \"${N2N_VERSION}\"
#define PACKAGE_BUILDDATE \"${N2N_VERSION}\"
#define HAVE_CONFIG_H 1
#define _WIN32_WINNT 0x0600
#define WINVER 0x0600
#define _CRT_SECURE_NO_WARNINGS 1
#define N2N_HAVE_WINTUN 1
")
endif()
file(WRITE ${CMAKE_BINARY_DIR}/include/config.h "${N2N_CONFIG_H}")

# Platform-specific settings
if(WIN32)
    add_definitions(-D_WIN32_WINNT=0x0600)
    add_definitions(-DWINVER=0x0600)
    add_definitions(-D_CRT_SECURE_NO_WARNINGS)
    add_definitions(-D_CONSOLE)
    add_definitions(-DGNUC=1)

    # Windows requires ws2_32 for networking
    set(PLATFORM_LIBS ws2_32 iphlpapi advapi32 kernel32 user32)

    # MinGW-specific settings
    if(MINGW)
        set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall")
        set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static")
    endif()
else()
    # POSIX systems
    set(PLATFORM_LIBS m pthread)

    # Linux capabilities
    find_library(CAP_LIBRARY NAMES cap)
    if(CAP_LIBRARY)
        set(PLATFORM_LIBS ${PLATFORM_LIBS} ${CAP_LIBRARY})
        add_definitions(-DN2N_HAVE_CAPABILITY=1)
    endif()
endif()

# OpenSSL support
if(N2N_ENABLE_OPENSSL)
    find_package(OpenSSL QUIET)
    if(OPENSSL_FOUND)
        include_directories(${OPENSSL_INCLUDE_DIR})
        set(PLATFORM_LIBS ${PLATFORM_LIBS} ${OPENSSL_LIBRARIES})
        add_definitions(-DN2N_HAVE_OPENSSL=1)
        message(STATUS "OpenSSL support: enabled")
    else()
        message(STATUS "OpenSSL support: disabled (not found)")
        set(N2N_ENABLE_OPENSSL OFF)
    endif()
endif()

# Zstd compression support
if(N2N_ENABLE_ZSTD)
    find_path(ZSTD_INCLUDE_DIR NAMES zstd.h)
    find_library(ZSTD_LIBRARY NAMES libzstd zstd)
    if(ZSTD_INCLUDE_DIR AND ZSTD_LIBRARY)
        include_directories(${ZSTD_INCLUDE_DIR})
        set(PLATFORM_LIBS ${PLATFORM_LIBS} ${ZSTD_LIBRARY})
        add_definitions(-DN2N_HAVE_ZSTD=1)
        message(STATUS "Zstd compression: enabled")
    else()
        message(STATUS "Zstd compression: disabled (not found)")
        set(N2N_ENABLE_ZSTD OFF)
    endif()
endif()

# Create n2n static library
add_library(n2n STATIC ${N2N_LIB_SOURCES})
target_link_libraries(n2n ${PLATFORM_LIBS})
set_target_properties(n2n PROPERTIES
    OUTPUT_NAME n2n
    VERSION ${N2N_VERSION}
    SOVERSION ${N2N_VERSION_MAJOR}
)

# edge executable
add_executable(edge WIN32 src/edge.c)
target_link_libraries(edge n2n ${PLATFORM_LIBS})

# supernode executable
add_executable(supernode src/supernode.c)
target_link_libraries(supernode n2n ${PLATFORM_LIBS})

# Install targets
install(TARGETS edge supernode n2n
    RUNTIME DESTINATION sbin
    ARCHIVE DESTINATION lib
    LIBRARY DESTINATION lib
)

# Windows-specific install for wintun.dll
if(WIN32 AND N2N_ENABLE_WINTUN)
    install(FILES ${CMAKE_SOURCE_DIR}/thirdparty/wintun.dll
        DESTINATION bin
        OPTIONAL
    )
endif()

# Test tools
if(N2N_ENABLE_TESTS)
    enable_testing()

    # Test executables from tools directory
    add_executable(tests-auth tools/tests-auth.c)
    target_link_libraries(tests-auth n2n ${PLATFORM_LIBS})

    add_executable(tests-compress tools/tests-compress.c)
    target_link_libraries(tests-compress n2n ${PLATFORM_LIBS})

    add_executable(tests-elliptic tools/tests-elliptic.c)
    target_link_libraries(tests-elliptic n2n ${PLATFORM_LIBS})

    add_executable(tests-hashing tools/tests-hashing.c)
    target_link_libraries(tests-hashing n2n ${PLATFORM_LIBS})

    add_executable(tests-transform tools/tests-transform.c)
    target_link_libraries(tests-transform n2n ${PLATFORM_LIBS})

    add_executable(tests-wire tools/tests-wire.c)
    target_link_libraries(tests-wire n2n ${PLATFORM_LIBS})

    add_test(NAME auth COMMAND tests-auth)
    add_test(NAME compress COMMAND tests-compress)
    add_test(NAME elliptic COMMAND tests-elliptic)
    add_test(NAME hashing COMMAND tests-hashing)
    add_test(NAME transform COMMAND tests-transform)
    add_test(NAME wire COMMAND tests-wire)
endif()

# Print configuration summary
message(STATUS "")
message(STATUS "=== n2n Configuration Summary ===")
message(STATUS "Version: ${N2N_VERSION}")
message(STATUS "Platform: ${CMAKE_SYSTEM_NAME}")
message(STATUS "C Compiler: ${CMAKE_C_COMPILER_ID} ${CMAKE_C_COMPILER_VERSION}")
message(STATUS "Install prefix: ${CMAKE_INSTALL_PREFIX}")
message(STATUS "OpenSSL support: ${N2N_ENABLE_OPENSSL}")
message(STATUS "Zstd compression: ${N2N_ENABLE_ZSTD}")
message(STATUS "Wintun support: ${N2N_ENABLE_WINTUN}")
message(STATUS "Tests: ${N2N_ENABLE_TESTS}")
message(STATUS "")
